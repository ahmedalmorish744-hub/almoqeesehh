    85→function recordToForm(r: any): LeaveFormData {

    87→    patient_name_ar: r.nameAr || "",
    88→    patient_name_en: r.nameEn || "",
    89→    id_number: r.identityNumber || "",
    90→    nationality_ar: r.nationalityAr || "",
    91→    nationality_en: r.nationalityEn || "",

    93→    employer_en: r.employerEn || "",
    94→    doctor_name_ar: r.doctorNameAr || "",
    95→    doctor_name_en: r.doctorNameEn || "",
    96→    position_ar: r.doctorSpecialtyAr || "",
    97→    position_en: r.doctorSpecialtyEn || "",
    98→    admission_date_gregorian: toDateInputValue(r.dateFrom) || r.dateFrom || "",
    99→    discharge_date_gregorian: toDateInputValue(r.dateTo) || r.dateTo || "",
   100→    hospital_name_ar: r.hospitalNameAr || "",
   101→    hospital_name_en: r.hospitalNameEn || "",
   102→    license_number: r.licenseNumber || "",
   103→    time: toTimeInputValue(r.timeFrom) || r.timeFrom || "",
   104→  };
   105→}
