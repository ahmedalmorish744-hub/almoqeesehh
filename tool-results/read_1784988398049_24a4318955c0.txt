   105→    def normalize_date_to_ddmmyyyy(self, date_str):
   106→        """تحويل أي صيغة تاريخ إلى DD-MM-YYYY
   107→        يدعم: DD-MM-YYYY, YYYY-MM-DD, DD/MM/YYYY, YYYY/MM/DD
   108→        """
   109→        if not date_str:
   110→            return date_str
   111→        try:
   112→            for sep in ['-', '/', '.']:
   113→                if sep in date_str:
   114→                    parts = date_str.split(sep)
   115→                    if len(parts) == 3:
   116→                        if len(parts[0]) == 4:
   117→                            # YYYY-MM-DD -> DD-MM-YYYY
   118→                            return f"{parts[2]}-{parts[1]}-{parts[0]}"
   119→                        else:
   120→                            # Already DD-MM-YYYY, normalize separator
   121→                            return f"{parts[0]}-{parts[1]}-{parts[2]}"
   122→            return date_str
   123→        except Exception:
   124→            return date_str

   126→    def parse_date_components(self, date_str):
   127→        """تحليل التاريخ وإرجاع (day, month, year) بغض النظر عن الصيغة"""
   128→        if not date_str:
   129→            return None
   130→        try:
   131→            for sep in ['-', '/', '.']:
   132→                if sep in date_str:
   133→                    parts = date_str.split(sep)
   134→                    if len(parts) == 3:
   135→                        if len(parts[0]) == 4:
   136→                            # YYYY-MM-DD
   137→                            return (int(parts[2]), int(parts[1]), int(parts[0]))
   138→                        else:
   139→                            # DD-MM-YYYY
   140→                            return (int(parts[0]), int(parts[1]), int(parts[2]))
   141→            return None
   142→        except Exception:
   143→            return None

   145→    def generate_leave_id(self, id_number, admission_date, discharge_date):
   146→        """توليد رمز الإجازة GSL فريد لكل إصدار، حتى لو كانت البيانات متطابقة.
   147→        البادئة الإجبارية: GSL (بصرف النظر عن نوع المنشأة)
   148→        النظام يدعم توليد أكثر من معرف إجازة فريد - كل محادثة بوت/كل سجل جديد يولد GSL مختلف تماماً.
   149→
   150→        الآلية:
   151→        - تُستخدم آخر 4 أرقام من رقم الهوية + أرقام من تاريخ الدخول + أرقام من تاريخ الخروج كأساس منطقي.
   152→        - يُضاف جزء عشوائي (4 أرقام) + طابع زمني (آخر 3 أرقام من الملي ثانية) لضمان أن كل رمز
   153→          جديد يختلف تماماً عن سابقه، حتى لو أُدخلت نفس البيانات بالضبط.
   154→        - الطول النهائي: 11 رقماً بعد البادئة GSL.
   155→        """
   156→        try:
   157→            # Normalize dates to DD-MM-YYYY first for consistent digit extraction
   158→            admission_normalized = self.normalize_date_to_ddmmyyyy(admission_date)
   159→            discharge_normalized = self.normalize_date_to_ddmmyyyy(discharge_date)
   160→
   161→            id_part = id_number[-4:] if len(id_number) >= 4 else (id_number or '0000')
   162→            admission_nums = ''.join(filter(str.isdigit, admission_normalized))[-3:].ljust(3, '0')
   163→            discharge_nums = ''.join(filter(str.isdigit, discharge_normalized))[-4:].ljust(4, '0')
   164→
   165→            # ✅ جزء عشوائي + طابع زمني لضمان التفرّد التام لكل إصدار
   166→            random_part = f"{random.randint(1000, 9999)}"
   167→            time_part = f"{int(time.time() * 1000) % 1000:03d}"
   168→            unique_part = random_part + time_part  # 4 + 3 = 7 أرقام
   169→
   170→            base = (discharge_nums + admission_nums + id_part)[:11].ljust(11, '0')
   171→            # استبدال آخر 7 أرقام بجزء التفرّد لضمان اختلاف الرمز لكل إصدار
   172→            leave_number = (base[:4] + unique_part)[:11].ljust(11, '0')
   173→            return f"GSL{leave_number}"
   174→        except Exception as e:
   175→            print(f"خطأ في توليد رمز الإجازة: {e}")
   176→            # حتى في حالة الخطأ، نُولّد رمزاً عشوائياً فريداً
   177→            random_part = f"{random.randint(10000000, 99999999)}"
   178→            return f"GSL260{random_part}"

   180→    def swap_date_format(self, date_str):
   181→        """تحويل التاريخ من YYYY-MM-DD إلى DD-MM-YYYY والعكس"""
   182→        if not date_str:
   183→            return date_str
   184→        try:
   185→            # First normalize to DD-MM-YYYY
   186→            normalized = self.normalize_date_to_ddmmyyyy(date_str)
   187→            # If already normalized (same as input), swap to the other format
   188→            parts = normalized.split('-')
   189→            if len(parts) == 3:
   190→                return f"{parts[2]}-{parts[1]}-{parts[0]}"
   191→            return date_str
   192→        except Exception:
   193→            return date_str

   195→    def calculate_duration(self, admission_date_hijri, discharge_date_hijri,
   196→                          admission_date_gregorian, discharge_date_gregorian):
   197→        """حساب مدة الإجازة - جميع التواريخ بصيغة DD-MM-YYYY (مثل 09-02-2026)
   198→        ملاحظة: نستخدم علامة LRM (U+200E) حول التواريخ لمنع خوارزمية BiDi
   199→        من عكس اتجاه الأرقام داخل السياق العربي (RTL).
   200→        """
   201→        # علامة Left-to-Right Mark لإجبار الأرقام على البقاء بترتيب LTR
   202→        LRM = '\u200e'
   203→        try:
   204→            admission_parsed = self.parse_date_components(admission_date_gregorian)
   205→            discharge_parsed = self.parse_date_components(discharge_date_gregorian)
   206→            # تطبيع التواريخ الميلادية إلى DD-MM-YYYY مسبقاً
   207→            admission_gregorian_normalized = self.normalize_date_to_ddmmyyyy(admission_date_gregorian)
   208→            discharge_gregorian_normalized = self.normalize_date_to_ddmmyyyy(discharge_date_gregorian)
   209→            # لف التواريخ بعلامات LRM لإجبارها على البقاء بصيغة DD-MM-YYYY في السياق العربي
   210→            admission_lrm = f"{LRM}{admission_gregorian_normalized}{LRM}"
   211→            discharge_lrm = f"{LRM}{discharge_gregorian_normalized}{LRM}"
   212→            if admission_parsed and discharge_parsed:
   213→                admission_dt = datetime(admission_parsed[2], admission_parsed[1], admission_parsed[0])
   214→                discharge_dt = datetime(discharge_parsed[2], discharge_parsed[1], discharge_parsed[0])
   215→                duration_days = (discharge_dt - admission_dt).days + 1
   216→
   217→                # صيغة DD-MM-YYYY للمدة العربية (مع علامات LRM لمنع العكس)
   218→                duration_ar = f"{duration_days} يوم  ( {admission_lrm} إلى {discharge_lrm} ) "
   219→
   220→                day_word = "day" if duration_days == 1 else "days"
   221→                # صيغة DD-MM-YYYY للمدة الإنجليزية أيضاً
   222→                duration_en = f"{duration_days} {day_word}  ( {admission_gregorian_normalized} to {discharge_gregorian_normalized} ) "
   223→                return duration_ar, duration_en
   224→            else:
   225→                duration_ar = f"1 يوم  ( {admission_lrm} إلى {discharge_lrm} ) "
   226→                duration_en = f"1 day  ( {admission_gregorian_normalized} to {discharge_gregorian_normalized} ) "
   227→                return duration_ar, duration_en
   228→        except Exception as e:
   229→            print(f"خطأ في حساب المدة: {e}")
   230→            duration_ar = f"1 يوم  ( {admission_lrm} إلى {discharge_lrm} ) "
   231→            duration_en = f"1 day  ( {admission_gregorian_normalized} to {discharge_gregorian_normalized} ) "
   232→            return duration_ar, duration_en

   375→    def has_arabic_chars(self, text):
   376→        """فحص هل النص يحتوي على حروف عربية"""
   377→        if not text:
   378→            return False
   379→        for char in text:
   380→            if '\u0600' <= char <= '\u06FF' or '\u0750' <= char <= '\u077F' or '\uFB50' <= char <= '\uFDFF' or '\uFE70' <= char <= '\uFEFF':
   381→                return True
   382→        return False

   384→    def is_arabic_char(self, char):
   385→        """فحص هل الحرف عربي"""
   386→        return ('\u0600' <= char <= '\u06FF' or
   387→                '\u0750' <= char <= '\u077F' or
   388→                '\uFB50' <= char <= '\uFDFF' or
   389→                '\uFE70' <= char <= '\uFEFF')

   391→    def split_mixed_text(self, text):
   392→        """تقسيم النص إلى أجزاء عربية وأجزاء أرقام/رموز"""
   393→        if not text:
   394→            return []
   395→        segments = []
   396→        current = ""
   397→        current_is_arabic = None
   398→        for char in text:
   399→            is_ar = self.is_arabic_char(char)
   400→            if current_is_arabic is None:
   401→                current_is_arabic = is_ar
   402→                current = char
   403→            elif is_ar == current_is_arabic:
   404→                current += char
   405→            else:
   406→                segments.append((current_is_arabic, current))
   407→                current_is_arabic = is_ar
   408→                current = char
   409→        if current:
   410→            segments.append((current_is_arabic, current))
   411→        return segments

   413→    def render_long_name_cell(self, x, y, width, height, text, row_idx, col_idx):
   414→        """عرض خلية الاسم/الطبيب الطويل بسطرين مع توسيط عمودي وأفقي"""
   415→        if not text:
   416→            return
   417→        # تحديد الخط والحجم
   418→        self.set_cell_font_and_color(row_idx, col_idx, text)
   419→        font_name = self.font_family
   420→        font_style = self.font_style
   421→        font_size = self.font_size_pt
   422→
   423→        text_width = self.get_string_width(text)
   424→        padding = 4  # مسافة أمان من الجوانب
   425→        available_width = width - padding * 2
   426→
   427→        if text_width <= available_width:
   428→            # النص يكفي بسطر واحد - عرض عادي بالتوسيط
   429→            self.set_xy(x, y)
   430→            self.cell(width, height, text, align='C')
   431→            return
   432→
   433→        # النص طويل - تقسيمه إلى سطرين
   434→        # البحث عن نقطة القطع المثالية: آخر مسافة قبل المنتصف أو بعد المنتصف
   435→        words = text.split(' ')
   436→        line1 = ''
   437→        line2 = ''
   438→        found_split = False
   439→
   440→        for i in range(len(words), 0, -1):
   441→            test_line = ' '.join(words[:i])
   442→            if self.get_string_width(test_line) <= available_width:
   443→                line1 = test_line
   444→                line2 = ' '.join(words[i:])
   445→                # إذا السطر الثاني طويل جداً، نقطع من منتصف الكلمة الأخيرة
   446→                if self.get_string_width(line2) > available_width and line2:
   447→                    # قطع الكلمة الأخيرة في السطر الأول بشرطة
   448→                    last_word = words[i-1] if i > 0 else ''
   449→                    # نحاول نقطع الكلمة الأخيرة
   450→                    for cut in range(len(last_word)-1, 0, -1):
   451→                        partial = last_word[:cut] + '-'
   452→                        test_line1 = ' '.join(words[:i-1] + [partial]) if i > 1 else partial
   453→                        remainder = last_word[cut:] + ' ' + ' '.join(words[i:]) if i < len(words) else last_word[cut:]
   454→                        if self.get_string_width(test_line1) <= available_width and self.get_string_width(remainder) <= available_width:
   455→                            line1 = test_line1
   456→                            line2 = remainder
   457→                            break
   458→                found_split = True
   459→                break
   460→
   461→        if not found_split or not line2:
   462→            # لا نجد نقطة قطع جيدة - عرض بسطر واحد
   463→            self.set_xy(x, y)
   464→            self.cell(width, height, text, align='C')
   465→            return
   466→
   467→        # حساب ارتفاع السطر وتوسيط عمودي
   468→        line_height = height / 2
   469→        # توسيط عمودي: حساب الإزاحة من الأعلى
   470→        total_text_height = line_height * 2
   471→        y_offset = y + (height - total_text_height) / 2
   472→
   473→        # إعادة تعيين الخط (لأن get_string_width قد تغيره)
   474→        self.set_font(font_name, font_style, size=font_size)
   475→
   476→        # السطر الأول
   477→        self.set_xy(x, y_offset)
   478→        self.cell(width, line_height, line1, align='C')
   479→
   480→        # السطر الثاني
   481→        self.set_xy(x, y_offset + line_height)
   482→        self.cell(width, line_height, line2, align='C')
   483→
   484→    def render_mixed_font_cell_v2(self, x, y, width, height, text, color):
   485→        """عرض خلية بخطين مع الحفاظ على ترتيب BiDi:
   486→        الأرقام والأقواس بخط Times، الحروف العربية بخط NotoSansArabic
   487→        يتجاهل علامات التحكم مثل LRM (U+200E) لأنها غير مرئية ولا تدعمها خط Times"""
   488→        if not text:
   489→            return
   490→        # إزالة علامات التحكم غير المرئية (LRM, RLM, ZWJ, ZWNJ) لأنها لا تدعمها خط Times
   491→        # وهي فقط لتوجيه BiDi ولا تظهر بصرياً
   492→        import unicodedata
   493→        clean_text = ''.join(
   494→            char for char in text
   495→            if unicodedata.category(char) != 'Cf'  # skip Format characters like LRM
   496→        )
   497→        text = clean_text
   498→        # حساب العرض الكلي لكل حرف بالخط المناسب
   499→        total_width = 0
   500→        for char in text:
   501→            if self.is_arabic_char(char):
   502→                self.set_font('NotoSansArabic-Regular', size=13)
   503→            else:
   504→                self.set_font('Times', '', size=13)
   505→            total_width += self.get_string_width(char)
   506→        # توسيط أفقي
   507→        start_x = x + (width - total_width) / 2
   508→        self.set_xy(start_x, y)
   509→        self.set_text_color(*color)
   510→        # عرض كل حرف بالخط المناسب باستخدام write للحفاظ على الترتيب
   511→        for char in text:
   512→            if self.is_arabic_char(char):
   513→                self.set_font('NotoSansArabic-Regular', size=13)
   514→            else:
   515→                self.set_font('Times', '', size=13)
   516→            self.set_text_color(*color)
   517→            self.write(height, char)
