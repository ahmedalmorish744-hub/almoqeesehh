    87→function recordToForm(r: any): LeaveFormData {
    88→  return {
    89→    patient_name_ar: r.nameAr || "",
    90→    patient_name_en: r.nameEn || "",
    91→    id_number: r.identityNumber || "",
    92→    nationality_ar: r.nationalityAr || "",
    93→    nationality_en: r.nationalityEn || "",
    94→    employer_ar: r.employer || "",
    95→    employer_en: r.employerEn || "",
    96→    doctor_name_ar: r.doctorNameAr || "",
    97→    doctor_name_en: r.doctorNameEn || "",
    98→    position_ar: r.doctorSpecialtyAr || "",
    99→    position_en: r.doctorSpecialtyEn || "",
   100→    admission_date_gregorian: toDateInputValue(r.dateFrom) || r.dateFrom || "",
   101→    discharge_date_gregorian: toDateInputValue(r.dateTo) || r.dateTo || "",
   102→    hospital_name_ar: r.hospitalNameAr || "",
   103→    hospital_name_en: r.hospitalNameEn || "",
   104→    license_number: r.licenseNumber || "",
   105→    time: toTimeInputValue(r.timeFrom) || r.timeFrom || "",
   106→  };
   107→}

   961→function StatusBlock({

   963→  status,
   964→  message,
   965→  icon,
   966→  onAction,
   967→  extra,
   968→}: {
