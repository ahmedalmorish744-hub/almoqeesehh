   188→// --- Helper for Isolation ---
   189→// If admin, can see all (OR just their own? User said "Admin control all components").
   190→// I will interpret "control all" as seeing everything.
   191→const applyIsolation = (query, params, user) => {
   192→    if (user.role === 'admin') return { query, params };
   193→
   194→    // If query has WHERE, add AND user_id = ?
   195→    // If not, add WHERE user_id = ?
   196→    if (query.toLowerCase().includes('where')) {
   197→        return { query: query + ' AND user_id = ?', params: [...params, user.id] };
   198→    } else {
   199→        // Handle ORDER BY or LIMIT which might be at the end
   200→        // Simple heuristic: insert WHERE before ORDER BY or LIMIT if they exist, or at end
   201→        const upperQ = query.toUpperCase();
   202→        const orderIdx = upperQ.indexOf('ORDER BY');
   203→        const limitIdx = upperQ.indexOf('LIMIT');
   204→        const cutIdx = (orderIdx !== -1 && limitIdx !== -1) ? Math.min(orderIdx, limitIdx)
   205→            : (orderIdx !== -1 ? orderIdx : limitIdx);
   206→
   207→        if (cutIdx !== -1) {
   208→            return {
   209→                query: query.substring(0, cutIdx) + ' WHERE user_id = ? ' + query.substring(cutIdx),
   210→                params: [user.id, ...params]
   211→            };
   212→        } else {
   213→            return { query: query + ' WHERE user_id = ?', params: [...params, user.id] };
   214→        }
   215→    }
   216→};

   840→// --- Database Auto-Setup on Startup ---
   841→async function waitForDatabase(maxRetries = 15, delayMs = 3000) {
   842→    for (let i = 0; i < maxRetries; i++) {
   843→        try {
   844→            await db.query('SELECT 1');
   845→            console.log('Database connection established.');
   846→            return true;
   847→        } catch (err) {
   848→            console.log(`Waiting for database... (${i + 1}/${maxRetries})`);
   849→            await new Promise(resolve => setTimeout(resolve, delayMs));
   850→        }
   851→    }
   852→    throw new Error('Could not connect to database after retries.');
   853→}

   855→async function initializeDatabase() {

   866→
   867→            for (const statement of statements) {

   869→                    await db.query(statement);
